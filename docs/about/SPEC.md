## 3. Схема данных

Порядок создания таблиц обязателен: справочники → проекты → документы → разделы → чанки → артефакты → рёбра. Циклических FK нет.

```sql
CREATE EXTENSION IF NOT EXISTS vector;

-- ---------- реестр версий индекса ----------
CREATE TABLE index_versions (
  version         text PRIMARY KEY,
  embedder_model  text NOT NULL,
  embedder_revision text,
  parser_version  text NOT NULL,
  chunk_params    jsonb NOT NULL,        -- CHUNK_TARGET, MAX, MIN, OVERLAP
  enrich_prompt_version text NOT NULL,
  created_at      timestamptz NOT NULL DEFAULT now(),
  status          text NOT NULL CHECK (status IN ('building','active','retired','failed'))
);
-- Одновременно активна ровно одна версия. Все поисковые запросы
-- фильтруются по index_version = активной. Предыдущая версия хранится
-- до подтверждения качества новой (регрессия на датасете), затем retired.

-- ---------- БД подтверждений ----------
CREATE TABLE clients (
  client_id      uuid PRIMARY KEY,
  canonical_name text NOT NULL,
  aliases        text[] NOT NULL DEFAULT '{}'
);

CREATE TABLE projects (
  project_id     uuid PRIMARY KEY,
  name           text NOT NULL,
  client_id      uuid REFERENCES clients(client_id),
  industry       text,
  publicity      text NOT NULL CHECK (publicity IN ('open','anon','closed')),
  anon_label     text,
  tender_result  text CHECK (tender_result IN ('won','lost','not_submitted')),
  amount         numeric,
  date_start     date, date_end date,
  company_role   text CHECK (company_role IN ('general','subcontractor','consultant')),
  tech_stack     text[],
  contract_no    text,
  status         text CHECK (status IN ('active','archived')),
  CONSTRAINT anon_label_required CHECK (publicity <> 'anon' OR anon_label IS NOT NULL)
);

-- Версионируемые загрузки реестра из Excel (раздел 4.5)
CREATE TABLE registry_uploads (
  upload_id    uuid PRIMARY KEY,
  loaded_at    timestamptz NOT NULL DEFAULT now(),
  source_file  text NOT NULL,
  row_count    int,
  rejected_count int,
  report       jsonb                      -- построчные ошибки валидации
);

-- ---------- корпус ----------
CREATE TABLE documents (
  doc_id        uuid PRIMARY KEY,
  project_id    uuid NOT NULL REFERENCES projects(project_id),
  title         text NOT NULL,            -- человекочитаемое название для списка «Источники»
  file_path     text NOT NULL UNIQUE,
  file_hash     text NOT NULL,
  doc_type      text NOT NULL CHECK (doc_type IN ('kp','report','act','contract','letter','presentation')),
  doc_date      date,                     -- дата документа, для boost по свежести
  status        text NOT NULL DEFAULT 'pending'
                CHECK (status IN ('pending','parsed','enriched','indexed','failed')),
  last_error    text,
  attempts      int NOT NULL DEFAULT 0,
  parsed_at     timestamptz,
  parser_version text,
  page_count    int
);
CREATE INDEX ON documents (file_hash);
-- Сопоставление файл → проект выполняется по реестру ДО индексации (шаг 0
-- офлайн-конвейера). Файл без сопоставленного проекта в индекс не попадает
-- и остаётся в статусе failed с внятной причиной: project_id NOT NULL
-- гарантирует это на уровне схемы. Поведение по умолчанию — «считать закрытым».

CREATE TABLE sections (
  section_id    uuid PRIMARY KEY,
  doc_id        uuid NOT NULL REFERENCES documents(doc_id) ON DELETE CASCADE,
  project_id    uuid NOT NULL REFERENCES projects(project_id),
  title         text NOT NULL,
  level         int  NOT NULL,
  page_start    int, page_end int,        -- NULL допустим: DOCX/PPTX без рендеринга страниц
  text          text NOT NULL,
  content_hash  text NOT NULL,            -- для инкрементальной переиндексации
  embedding     vector(1024) NOT NULL,
  token_count   int,
  index_version text NOT NULL REFERENCES index_versions(version)
);
CREATE INDEX ON sections (doc_id);
CREATE INDEX ON sections (index_version);

CREATE TABLE chunks (
  chunk_id        uuid PRIMARY KEY,
  doc_id          uuid NOT NULL REFERENCES documents(doc_id) ON DELETE CASCADE,
  project_id      uuid NOT NULL REFERENCES projects(project_id),
  ord             int  NOT NULL,
  section_title   text,
  subsection_title text,
  parent_section_id uuid REFERENCES sections(section_id),
  page_start      int, page_end int,      -- NULL допустим (см. sections)
  text_raw        text NOT NULL,
  text_enriched   text NOT NULL,
  content_hash    text NOT NULL,          -- hash(text_raw), ключ кэша обогащения
  embedding       vector(1024) NOT NULL,  -- по text_enriched; используется в поиске
  embedding_raw   vector(1024) NOT NULL,  -- по text_raw; используется ТОЛЬКО для
                                          -- дедупликации, MMR и семантических рёбер.
                                          -- Обогащение добавляет заказчика и отрасль,
                                          -- поэтому близость копий типового блока по
                                          -- text_enriched падает ниже порога — дедупликация
                                          -- по нему не работает по построению
  tsv             tsvector GENERATED ALWAYS AS (to_tsvector('russian'::regconfig, text_raw)) STORED,
  token_count     int,
  is_table        boolean DEFAULT false,
  ocr_low_conf    boolean DEFAULT false,  -- потребитель: пометка «⚠ низкое качество
                                          -- распознавания» в списке «Источники»; из выдачи
                                          -- не исключается
  cluster_id      uuid,
  cluster_size    int  DEFAULT 1,         -- для флага boilerplate и метрики уникальности
  is_canonical    boolean DEFAULT true,
  index_version   text NOT NULL REFERENCES index_versions(version)
);
CREATE INDEX ON chunks USING gin(tsv);
CREATE INDEX ON chunks (project_id);
CREATE INDEX ON chunks (cluster_id, is_canonical);
CREATE INDEX ON chunks (index_version);
-- ANN-индекс НЕ создаётся: точный перебор по 20 тыс. векторов

CREATE TABLE artifacts (
  artifact_id  uuid PRIMARY KEY,
  project_id   uuid NOT NULL REFERENCES projects(project_id),
  artifact_type text NOT NULL CHECK (artifact_type IN ('act','contract','letter')),
  title        text NOT NULL,             -- наименование для списка «Подтверждения»
  number       text,
  issued_at    date,
  file_path    text NOT NULL,
  doc_id       uuid REFERENCES documents(doc_id)  -- связь с проиндексированной копией,
                                                  -- если файл артефакта входит в корпус
);

-- ---------- граф связей (MVP+) ----------
CREATE TABLE chunk_edges (
  src_chunk_id uuid REFERENCES chunks(chunk_id) ON DELETE CASCADE,
  dst_chunk_id uuid REFERENCES chunks(chunk_id) ON DELETE CASCADE,
  edge_type    text NOT NULL CHECK (edge_type IN ('parent','sibling','duplicate','semantic','project')),
  weight       real,
  PRIMARY KEY (src_chunk_id, dst_chunk_id, edge_type)
);
```

**Векторы и метрика.** Все эмбеддинги нормализуются при записи (L2=1). Оператор pgvector — косинусный `<=>`. Пороги `DEDUP_THRESHOLD` и `GRAPH_EDGE_THRESHOLD` определены на косинусной близости нормализованных векторов.

**Миграции.** Каталог `migrations/` с нумерованными SQL-файлами и простым раннером (либо Alembic — решение разработчика на этапе 0.5, фиксируется и не меняется). Ручные правки схемы на стенде запрещены. DDL выше — миграция 0001.

Трассы запросов хранятся в Langfuse, не в этой схеме.

## 4. Офлайн-конвейер (индексация)

Триггер: файл помещён в отслеживаемую папку либо изменился `file_hash`.

### 4.1 Контракт со слоем подготовки данных (часть 1)

Модуль индексации принимает на вход JSON на документ (схема фиксируется до этапа 2 и версионируется):

```
{ "source_file": str, "file_hash": str, "parser_version": str,
  "page_count": int | null,
  "sections": [ { "title": str, "level": int,
                  "page_start": int | null, "page_end": int | null,
                  "blocks": [ { "type": "paragraph" | "table" | "caption" | "header",
                                "text": str,
                                "page": int | null,
                                "ocr_conf": float | null } ] } ] }
```

Для DOCX/PPTX номера страниц берутся из рендеринга LibreOffice в PDF; при невозможности — `null`, ссылка в «Источниках» строится по разделу без страниц.

### 4.2 Шаги

| # | Шаг | Параметры | Контур |
|---|---|---|---|
| 0 | Сопоставление файла с проектом реестра | по пути/имени и правилам реестра; нет соответствия → документ в `failed`, в индекс не попадает («по умолчанию закрыт») | PoC |
| 1 | Классификация типа **страницы** (текст / скан) | постранично; определяет ветку парсинг/OCR. Тип документа (`doc_type`) берётся из реестра | часть 1 |
| 2 | Парсинг или OCR | порог: <100 символов на страницу → OCR | часть 1 |
| 3 | Выделение таблиц | таблица = неделимый чанк + подпись + предшествующий абзац; таблица длиннее `CHUNK_HARD_LIMIT_EMBED` режется по строкам с повторением шапки и подписи в каждой части | часть 1 / PoC |
| 4 | Чанкинг | цель 500, max 1000 (кроме таблиц), min 100, overlap 15% — только внутри раздела, с конца предыдущего чанка | PoC |
| 5 | Обогащение контекстом | см. 4.3 | PoC |
| 6 | Метаданные | см. схему `chunks` | PoC |
| 7 | Векторизация | `embedding` по `text_enriched`, `embedding_raw` по `text_raw`; секционный индекс по `sections.text`; батч 128 | PoC |
| 8 | Дедупликация | по `embedding_raw`, косинус > 0.95 → кластер (связная компонента; ручной просмотр 20 крупнейших кластеров при первой индексации, запасной порог 0.97); `is_canonical` у одного, `cluster_size` у всех | PoC |
| 9 | FTS-индекс | генерируется колонкой `tsv` автоматически | PoC |
| 10 | Граф связей | рёбра `parent`, `sibling`, `duplicate`, `project` тривиальны; `semantic` по `embedding_raw` при косинусе > 0.75–0.85 | MVP+ |

Дедупликация считается блочным матричным умножением (батчи по ~2000 векторов), не полной матрицей 20000×20000 в памяти.

Правила разбиения (шаг 4): раздел → подраздел → абзац → предложение, дробление только до попадания в границы. Соседние короткие подразделы склеиваются.

Раздел длиннее лимита эмбеддера (`SECTION_MAX_TOKENS`): запись секционного индекса разбивается на части с общим `section_id`-префиксом в `title` («Методология, ч. 1/3»); каждая часть ≤ `SECTION_MAX_TOKENS`.

### 4.3 Обогащение контекстом

- **Вход промпта:** текст чанка + метаданные из реестра и парсера (проект, заказчик с учётом публичности, год, отрасль, тип документа, путь раздела). Полный документ в промпт **не** подаётся.
- **Промпт версионируется** (`enrich_prompt_version` в `index_versions`), температура 0, ограничение ответа 120 токенов.
- **Кэш обязателен:** ключ `(content_hash, enrich_prompt_version)`, хранение в таблице `enrichment_cache(content_hash, prompt_version, enriched_prefix, created_at)`. Смена модели эмбеддингов или параметров векторизации **не** инвалидирует кэш — переиндексация без повторного обогащения занимает минуты. Смена чанкинга инвалидирует кэш частично (по совпадению `content_hash`).
- **Фолбэк:** при отказе/таймауте модели на чанке — детерминированный префикс из метаданных («Фрагмент из {тип} для {заказчик|anon_label}, {год}, раздел „{раздел}“»). Фолбэк помечается в записи кэша.
- **Бюджет:** первичное обогащение 20 тыс. чанков ≈ одна ночь GPU. Это самая дорогая операция конвейера; замер на 100 чанках — критерий этапа 2 до полного прогона.
- Качество обогащения проверяется ручным просмотром 50 случайных чанков (чек-лист: контекст фактически верен, публичность не нарушена) — до полного прогона.

### 4.4 Идемпотентность и транзакционность

- Одна транзакция на документ. Повторный запуск пропускает документы `status='indexed'` с совпадающим `file_hash`.
- Индексация пишет в **новую** `index_version` со статусом `building`; активная версия переключается атомарно после прохождения регрессии на эталонном датасете. Это же даёт откат: предыдущая версия остаётся `active` до подтверждения.
- Падение посреди корпуса: документы со статусом `indexed` не переделываются, `failed` перезапускаются с инкрементом `attempts`.
- Обновление документа: переиндексация разделов с изменившимся `content_hash`; полная переиндексация файла только при изменении структуры.

### 4.5 Загрузчик БД подтверждений

Отдельный модуль. Вход — Excel с закрытыми списками (выпадающие меню). Валидации при загрузке, строки с ошибками отклоняются с построчным отчётом в `registry_uploads.report`:

1. Обязательные поля заполнены.
2. Справочные значения (тип артефакта, отрасль, статус, результат тендера, публичность) — по закрытым спискам.
3. Ссылки на файлы существуют.
4. Названия заказчиков нормализованы по `clients.aliases`.
5. Каждая загрузка версионируется (`registry_uploads`).

## 5. Онлайн-конвейер (запрос)

### 5.1 Разбор запроса

| # | Шаг | Детали | Исполнитель |
|---|---|---|---|
| 1 | Классификация типа задачи | закрытый список (раздел 7), JSON-ответ, temperature 0 | GigaChat |
| 2 | Выбор шаблона плана | детерминированно по типу | код |
| 3 | Извлечение параметров | отрасль, тип задачи, ограничения; сверка со справочниками; JSON-ответ | GigaChat |
| 4 | Развёртывание в подзапросы | по граням подхода | Jinja2 |
| 5 | Генерация гипотез HyDE | по одной на грань, ≤150 токенов | GigaChat |

**Контракт LLM-вызовов (все):** формат ответа — JSON по схеме; temperature 0 (кроме HyDE: 0.7); до 2 повторов при невалидном JSON, затем детерминированный фолбэк. Фолбэк классификации — расширенный план со всеми гранями. «Уверенность» классификации — самооценка модели в JSON-ответе; при значении ниже `CLASSIFIER_CONF_THRESHOLD` также берётся расширенный план. (Самооценка квантизованной модели плохо откалибрована — метрика точности классификации на датасете обязана это учесть; калибровка порога по матрице ошибок.)

Грани подхода: `understanding`, `methodology`, `stages`, `artifacts`, `risks`, `outcome`. Состав задаётся шаблоном и различается по типам задач.

**Инвариант HyDE:** текст гипотезы не попадает в контекст генерации. Двойная защита: (1) тест, проверяющий отсутствие гипотезы в промпте генерации; (2) рантайм-ассерт перед отправкой промпта — при вхождении текста любой гипотезы в промпт запрос падает с записью инцидента в трассу. В трассу гипотезы пишутся полностью.

### 5.2 Поиск

На каждую грань:

```
ключи   = [подзапрос, гипотеза]
# 4 списка по PER_LIST_K=100 кандидатов каждый:
списки  = [vector(k, PER_LIST_K) for k in ключи] + [fts(k, PER_LIST_K) for k in ключи]
кандидаты = weighted_RRF(списки, k=60, weights={запрос:1.0, гипотеза:0.7})[:CANDIDATES=50]
          # веса гарантируют: гипотеза добавляет кандидатов, но не вытесняет
          # найденное исходным запросом (RRF без весов этого НЕ гарантирует)
ранжированные = rerank(подзапрос, кандидаты)        # вход реранкера — text_raw,
                                                    # усечение до лимита сервиса с конца
if max(score) < RERANK_THRESHOLD: грань = НЕ_НАЙДЕНО
boosted = apply_boost(ранжированные)                # см. ниже
финал   = MMR(boosted, lambda=0.7)[:TOP_K_FINAL=8]
```

- **Все запросы к `chunks`/`sections` включают:** `index_version = :active`, `is_canonical = true`, и фильтр публичности:
  `JOIN projects p ON p.project_id = c.project_id AND p.publicity = ANY(:ALLOWED_PUBLICITY)`.
  `ALLOWED_PUBLICITY = ['open']` на PoC (см. раздел 12; данные `anon` не мутируются).
- **FTS:** запрос строится как дизъюнкция (`|`) 8 наиболее редких лемм ключа (по IDF корпуса); ранжирование `ts_rank_cd`. `plainto_tsquery` от полного текста запрещён — конъюнкция всех лемм гипотезы на 150 слов даёт пустой список, и половина схемы поиска молча вырождается.
- **Boost** (после реранкинга, до MMR): `score' = score * (1 + w_ind*[отрасль совпала] + w_won*[tender_result='won'] + w_fresh*max(0, 1 - возраст_лет/5))`; веса в конфигурации, по умолчанию 0.05/0.05/0.03. Незаполненные метаданные дают 0, не штраф.
- **MMR:** оба слагаемых min-max-нормируются по множеству кандидатов (шкалы реранкера и косинуса несопоставимы — без нормировки MMR вырождается в порядок реранкера). Близость между чанками — по `embedding_raw`.
- **Провенанс:** кандидат — объект, несущий через весь конвейер список исходных списков и позиций (для метрики вклада HyDE) — фиксируется в трассе.

**Секционный индекс (обязательная ветвь).** Для типов задач со сценарием генерации по образцу параллельно чанковому поиску: векторный поиск по `sections` ключом исходного запроса (без HyDE) → реранкинг топ-10 → отбор `SECTION_EXEMPLARS = 2` разделов выше `RERANK_THRESHOLD`. Разделы-образцы — основной материал контекста генерации; чанки и подтверждения дополняют. Без этой ветви система вырождается в вопросно-ответную, от чего концепция v2.5 отказалась явно.

**Поведение при НЕ_НАЙДЕНО:** генерация продолжается по найденным граням; непокрытые грани перечисляются пользователю явным блоком «Не найдено в корпусе: …». Если не найдены **все** грани или обязательная грань шаблона — отказ целиком, без генерации. Факт и состав отказа пишутся в трассу (метрика корректных отказов).

Параллельно: параметризованные запросы к БД подтверждений.

### 5.2а Параметризованные запросы к БД подтверждений

Реестр из 10–15 шаблонов (**не text2SQL**), формат — файл в git: `{id, sql, params_schema (JSON Schema), task_types[]}`. Модель выбирает id из закрытого списка и подставляет параметры; параметры валидируются по схеме и справочникам, невалидные → запрос не выполняется, факт в трассу. Фильтр `publicity = ANY(:ALLOWED_PUBLICITY)` применяется **и здесь** — включён в каждый шаблон. Пустая выдача — легальный результат, отражается в блоке «Подтверждения» как отсутствие.

### 5.3 Генерация

1. Сборка контекста: разделы-образцы (2) → факты БД подтверждений → чанки по граням. Дедупликация чанков между гранями по `chunk_id` и по `cluster_id`. Бюджет `CONTEXT_BUDGET_TOKENS = 12000` (токенизатор bge-m3, проверка токенизатором GigaChat ×1.2); при превышении усечение с хвоста списка чанков, образцы и подтверждения не усекаются. Для сценария тендерной заявки приоритет подтверждений над описанием подхода (концепция v2.5, сценарий А).
2. Подстановка шаблона раздела (структура, объём, стиль).
3. Генерация с обязательными маркерами `[N]`: фрагменты нумеруются в порядке следования в промпте; маркер с несуществующим номером удаляется на постобработке, утверждение помечается как непроверенное. Гранулярность проверки маркеров — предложение.
4. Два раздельных списка в конце:
   - `Источники` — `documents.title`, раздел, страницы (или раздел без страниц при NULL); канонический чанк разворачивается в перечень документов кластера («встречается также в: …»); пометка `⚠ OCR` при `ocr_low_conf`
   - `Подтверждения` — тип, `artifacts.title`, номер, дата, заказчик (через projects→clients, с учётом публичности), файл
5. Утверждения без маркера помечаются как непроверенные.

## 6. Внешний сервис: интеграция

### 6.1 Эмбеддинги

```http
POST /v1/embeddings
Authorization: Bearer <token>
{"model": "BAAI/bge-m3", "input": ["текст1", "текст2"], "encoding_format": "float"}

→ {"data": [{"index": 0, "embedding": [...1024 чисел...]}], "usage": {...}}
```

Разреженные векторы через этот интерфейс, скорее всего, недоступны — лексический поиск строится на PostgreSQL FTS.

### 6.2 Реранкинг — два возможных диалекта

```http
# Вариант A (TEI)
POST /rerank
{"query": "...", "texts": ["...", "..."], "raw_scores": false}
→ [{"index": 0, "score": 0.93}, ...]

# Вариант B (Cohere / Jina / vLLM)
POST /v1/rerank
{"model": "BAAI/bge-reranker-v2-m3", "query": "...", "documents": ["...", "..."]}
→ {"results": [{"index": 0, "relevance_score": 0.93}, ...]}
```

Диалект определяется прогоном `probe_api.py` до начала разработки. **Блокирующий вопрос:** сколько пар реранкер принимает за вызов — 50 кандидатов это 1 запрос или 50? При 6 гранях разница между 6 и 300 обращениями на запрос переворачивает расчёт латентности (раздел 13).

### 6.3 Доступность и деградация

Выделенного health-эндпоинта нет (уточняется probe_api). Синтетическая проба раз в 30 с: `GET /v1/models` либо `POST /v1/embeddings` с константной строкой.

Повторы: 3 попытки, задержки 1 / 3 / 9 с; при 429 учитывается `Retry-After`, если отдан. Кэш векторов запросов: ключ `(hash(текст), модель, ревизия модели)` — без ревизии кэш после обновления модели молча отдаёт несовместимые векторы. Гипотезы HyDE не кэшируются (недетерминированы).

**Отказ посреди запроса** (сервис упал на грани N из 6 после исчерпания повторов): запрос отклоняется целиком с сообщением «сервис поиска временно недоступен». Частичный ответ без части граней не выдаётся — для текста, идущего в тендерную заявку, немаркированная неполнота опаснее отказа.

### 6.4 Модуль изоляции

Обязателен, пишется на этапе 0.5. Единая точка обращения к эмбеддингам и реранкеру, две реализации (внешний сервис / локальный хостинг), переключение конфигурацией.

```python
class Embedder(Protocol):
    def embed(self, texts: list[str]) -> list[list[float]]: ...

class Reranker(Protocol):
    def rerank(self, query: str, docs: list[str]) -> list[tuple[int, float]]: ...
```

Требования к реализации: таймауты connect/read из конфигурации; батч ≤ `EMBED_BATCH`, вход усечён до лимита сервиса (значение — из probe_api); результат сортируется по полю `index`, порядок элементов ответа не считается гарантированным; параллелизм ≤ `EXTERNAL_MAX_CONCURRENCY` с учётом лимита 20 rps. Точки вызова: индексация, секционный индекс, обогащение (эмбеддинги не нужны, но модуль генерации изолируется аналогично), векторизация запроса, переиндексация. Прямых обращений к URL вне модуля нет — проверяется grep-тестом в CI.

## 8. Трассировка

Одна трасса на запрос, вложенные стадии. Обязательный состав:

- **Общее:** trace_id, время, **пользователь** (на PoC — выбор имени из списка при входе в Streamlit; без идентификации продуктовые метрики не считаются), исходный запрос; версии промптов, шаблонов, эмбеддера, реранкера, индекса (`index_version`), конфигурации (хэш собранной конфигурации)
- **Разбор:** тип задачи + уверенность, выбранный шаблон, параметры, подзапросы, **полный текст гипотез**
- **Поиск (на грань):** 4 исходных списка раздельно (chunk_id, score, позиция) → RRF-50 → результат реранкинга → boost → финальные после MMR → факт срабатывания порога; отбор секционных образцов
- **Генерация:** chunk_id/section_id контекста, текст, ссылки, утверждения без источника, срабатывания фильтра публичности и рантайм-ассерта HyDE
- **Тайминги:** по каждой стадии отдельно

Сбой трассировки не роняет пользовательский запрос (запись best-effort, ошибка — в лог приложения).

Обратная связь:
- явная — пометка «не по делу» рядом с каждым источником
- неявная — загрузка итогового документа (расстояние редактирования; связь с трассой — пользователь загружает документ со страницы конкретного ответа), переформулировка в течение минуты, отсутствие действий. Событие copy штатными средствами Streamlit не снимается — на PoC не собираем, вносим в MVP+ вместе с кастомным компонентом.

Langfuse разворачивается **до** начала измерений, на этапе 3. Офлайн-конвейер логирует стадии и ошибки в структурированный лог приложения с этапа 2 (Langfuse для офлайна не обязателен, но качество обогащения проверяется по 4.3).

## 9. Метрики и пороги

### Эталонный датасет

100–150 пар, формулировки **пользователя**, не термины документов. Разбиение 70/30. Обязательны запросы с заведомо отсутствующим ответом — **не менее 15** (по ним калибруется `RERANK_THRESHOLD`).

**Состав записи:** запрос → ожидаемый ответ → перечень эталонных **якорей**. Якорь = `(doc_id, диапазон страниц | путь раздела, цитата 1–2 предложения)`. Датасет **не** привязывается к `chunk_id`: чанки пересоздаются при каждом изменении параметров чанкинга, и разметка по chunk_id обнуляется — а сравнение вариантов чанкинга и есть главное назначение датасета. Резолвер сопоставляет якоря чанкам текущей `index_version` по вхождению цитаты (нормализованное сравнение); якоря, не разрешившиеся в текущем индексе, — отдельная метрика качества индексации.

**Трудоёмкость и владелец:** сбор формулировок — 12 РП по 8–10 запросов (2–3 часа суммарно); **разметка якорей — отдельная работа, оценка 3–5 рабочих дней человека, знающего корпус; исполнитель назначается на этапе 0 и это блокирующий пункт** (раздел 13). Разметка без якорей (только запрос-ответ) не позволяет измерять поиск детерминированно.

Датасет растёт из трасс до 300–500. Регрессия: сравнение **парное** на одних и тех же запросах, бутстрэп-интервал 95%; блокирует выкладку ухудшение Recall@K, значимое по парному критерию (а не точечные 3 п.п., которые на 100–150 запросах лежат в шуме).

### Retrieval (детерминированно, без LLM)

Полнота считается на уровне **кластера**: эталонный якорь засчитан, если в выдаче канонический представитель его кластера.

| Метрика | Определение | Цель |
|---|---|---|
| Recall@50 | доля якорей запроса, попавших в объединение кандидатов до реранкинга | ≥ 0.95 |
| Recall@K_ctx | доля якорей в **итоговом контексте генерации** при фиксированном бюджете `CONTEXT_BUDGET_TOKENS` | ≥ 0.85 |
| Recall@8 по грани | доля якорей грани в финальных 8 чанках грани | ≥ 0.80 |
| MRR | по первому релевантному в контексте | ≥ 0.7 |
| Доля уникальных кластеров в топ-8 | различных `cluster_id` среди финальных 8 | ≥ 0.75 |
| nDCG@8 | по грани | динамика |

**Сравнение этапов 3 и 3.5 — при равном бюджете контекста.** База (этап 3, без слоя разбора) и слой разбора (3.5) сравниваются по Recall@K_ctx на одинаковом `CONTEXT_BUDGET_TOKENS`, не по числу чанков: иначе прирост гарантирован арифметикой размера выдачи, и критерий «прирост положителен, иначе слой отключается» не способен сработать на отключение. Число якорей на запрос не ограничено, поэтому у широких запросов максимум Recall@K_ctx < 1 — цель 0.85 интерпретируется по медиане запросов, широкие запросы отслеживаются отдельно.

### Слой разбора запроса

- прирост Recall@K_ctx от декомпозиции (равный бюджет)
- прирост от HyDE (3 режима: запрос / гипотеза / объединение)
- доля финальных чанков, найденных **только** гипотезой (из провенанса, на живом трафике)
- точность классификации типа задачи (матрица ошибок)
- доля запросов, где HyDE ухудшил выдачу

### Generation

Faithfulness, точность цитирования (детерминированно: сверка маркеров с контекстом, гранулярность — предложение), answer relevance, доля корректных отказов.
**Нарушения публичности — строго 0** (детерминированная сверка выдачи со списком закрытых наименований и алиасов).

Судья на PoC — человек: 2 РП × 100–150 ответов **по фиксированному чек-листу** (артефакт этапа 5; 30 ответов оба РП оценивают пересекающимся набором для оценки согласованности). Автоматическая оценка — после калибровки по человеческой; судьёй не может быть GigaChat (самореферентность) — внешняя модель офлайн на обезличенной подвыборке.

### Продуктовые

Доля текста без правок (главная), время подготовки документа (базовая линия 3–5 дней), частота обращений, доля запросов без результата.

### Эксплуатационные

Латентность p50/p95 по стадиям (из трасс); полнота индексации; доля страниц на ручную проверку после OCR; задержка от загрузки документа до доступности в поиске.

## 10а. Структура проекта, тесты, ошибки

```
bzpo/
  ingest/        # контракт части 1, чанкинг, обогащение, дедупликация, индексация
  registry/      # загрузчик Excel → БД подтверждений, валидации
  storage/       # репозитории PostgreSQL; все SQL живут только здесь
  providers/     # модуль изоляции: Embedder/Reranker (external|local), клиент Ollama
  parsing/       # разбор запроса: классификация, параметры, HyDE, шаблоны планов
  search/        # vector+FTS, weighted-RRF, rerank, boost, MMR, секционные образцы
  generation/    # сборка контекста, маркеры, два списка, контроль публичности
  eval/          # резолвер якорей, прогон датасета (CLI), регрессионный гейт
  ui/            # Streamlit
  migrations/
```

Правила границ: обращения к БД — только из `storage/`, к внешнему сервису и Ollama — только из `providers/` (проверяется grep-тестом в CI).

**Обязательный минимум тестов:** RRF и weighted-RRF на известных списках; MMR с проверкой нормировки; чанкер на границах (min/max/overlap/неделимость таблицы); фильтр публичности (включая отсутствие проекта); валидатор маркеров цитирования; параметризованные запросы на фикстурной БД; модуль изоляции на моке (оба диалекта, 429, таймаут, битый JSON, перемешанный порядок `index`); идемпотентность индексации (двойной прогон — то же число чанков); инвариант HyDE (тест + рантайм-ассерт); соответствие пар «план ↔ разделы». Прогон эталонного датасета — отдельная CLI-команда: без неё регрессионный гейт раздела 9 не существует.

**Таксономия ошибок для пользователя** (Streamlit не должен показывать стектрейсы): сервис поиска недоступен; модель генерации недоступна; ничего не найдено (с перечнем ненайденных граней); ответ не прошёл валидацию формата; превышен бюджет времени. Каждому классу — фиксированное сообщение.

## 11. Латентность (расчёт, проверяется замером на этапе 3)

| Стадия | Оценка | Примечание |
|---|---|---|
| Классификация + параметры | 2–4 с | 2 вызова GigaChat |
| Гипотезы HyDE, 6 × ≤150 ток | 7–11 с | последовательно при Ollama по умолчанию; при выходе из бюджета — гипотезы параллельно (`num_parallel`, +KV-кэш) либо 1 гипотеза на 2 грани |
| Векторизация ключей | 0.3–1 с | внешний сервис, параллельно по граням |
| Поиск (вектор+FTS, 6 граней) | 0.1–0.5 с | PostgreSQL; требование — грани параллельно |
| Реранкинг 6×50 | 1.5–15 с | зависит от батча пар за вызов — блокирующий вопрос |
| Генерация раздела 800–2500 ток | 6–30 с | 80–130 ток/с — **оценка, проверяется пробой модели (раздел 13)** |
| **Итого** | **17–60 с** | впритык к бюджету; компенсаторы перечислены выше |

Ollama сериализует запросы: второй одновременный пользователь ждёт. Для 18 пользователей с пиковой параллельностью 2–3 на PoC приемлемо (очередь с индикацией в UI), фиксируется как известное ограничение.

## 12. Границы PoC и MVP+

**PoC:** чанкинг, метаданные, дедупликация (по `embedding_raw`), обогащение с кэшем, два индекса, гибридный поиск (weighted-RRF), реранкинг, boost, MMR (нормированный), порог отсечения, декомпозиция, HyDE с двумя ключами, секционные образцы в генерации, БД подтверждений с загрузчиком и параметризованными запросами, фильтрация публичности через `ALLOWED_PUBLICITY=['open']`, маркеры `[N]` и два списка, трассировка Langfuse, эталонный датасет на якорях.

**MVP+:** обезличивание и сверка результата (рубежи 2 и 3; тогда `ALLOWED_PUBLICITY=['open','anon']`), жёсткий фильтр по явному выбору пользователя, граф связей и расширение выдачи (хук в модуле поиска: после MMR, до сборки контекста, с реранкингом добранных), предварительные резюме, событие copy в UI, перенос эмбеддингов на своё оборудование.

**Отдельно:** LightRAG — самостоятельный эксперимент после PoC.

**Следствие для PoC:** рубежи 2 и 3 отсутствуют → любой исходящий документ вычитывается человеком; проекты `anon` исключаются из выдачи конфигурацией `ALLOWED_PUBLICITY` — **разметка `publicity` в данных не мутируется** (перевод anon→closed в данных необратимо терял бы разметку).

## 13. Блокирующее до начала разработки (этап 0)

1. Прогон `probe_api.py` → диалект реранкера, предел длины входа, батч пар реранкера за вызов, схема лимитов (по запросам или элементам).
2. **Проба GigaChat-20B-A3B:** q4_K_M существует и поднимается в Ollama, MoE поддержан, шаблон чата корректен, замер ток/с и максимального `num_ctx` на целевой карте. Полдня; без неё весь расчёт латентности и этап обогащения — предположение. Запасной вариант — Qwen3.5-27B.
3. Эталонный датасет: формулировки собраны (12 РП × 8–10), **якорная разметка назначена конкретному исполнителю со сроком** (3–5 дней работы).
4. Поле `tender_result` заведено и заполнено ретроспективно по 25 проектам.
5. `publicity` проставлен по всем проектам; **посчитано распределение** open/anon/closed — если после фильтра `['open']` остаётся < 60% корпуса, решение о жизнеспособности PoC пересматривается до старта.
6. Решение о закупке карты 24 ГБ принято, владелец и срок зафиксированы.
7. Разведка корпуса (часть 1, этап 1) — может изменить оценку сроков.

## 15. Открытые вопросы к владельцу внешнего сервиса

Блокирующие (этап 0): спецификация OpenAPI/Swagger; батч пар реранкера за вызов; лимит 20 rps — по запросам или элементам; реальная доступность 8192 токенов контекста; наличие health-эндпоинта; точные ревизии развёрнутых моделей.

Эксплуатационные: сервисный токен вместо персонального; уведомление о смене версии модели; поведение при превышении лимита (429/таймаут/очередь); предупреждение о перезапусках; замер 214 чанков/с — при свободной карте или под нагрузкой.

Организационные (вне сервиса): владелец БД подтверждений на период эксплуатации; ежеквартальная сверка реестра; ответственный за добавление документов после PoC.
